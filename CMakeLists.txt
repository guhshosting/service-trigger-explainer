cmake_minimum_required(VERSION 3.24)
project(service-trigger-explainer VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows and MSVC are required")
endif()
option(STE_ANALYZE "Enable MSVC static analysis" OFF)
option(STE_ASAN "Enable AddressSanitizer" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ste_core src/model.cpp src/guid_db.cpp src/data.cpp src/decode.cpp src/json.cpp src/human.cpp src/error.cpp src/scm.cpp src/cli.cpp)
target_include_directories(ste_core PUBLIC include)
target_compile_definitions(ste_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(ste_core PUBLIC advapi32 ole32)
add_executable(service-trigger-explainer src/main.cpp src/version.rc)
target_link_libraries(service-trigger-explainer PRIVATE ste_core)
target_link_options(service-trigger-explainer PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
enable_testing()
add_executable(ste_tests tests/tests.cpp)
target_link_libraries(ste_tests PRIVATE ste_core)
foreach(t ste_core service-trigger-explainer ste_tests)
  target_compile_options(${t} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/utf-8;/permissive-;/EHsc>)
  if(STE_ANALYZE)
    target_compile_options(${t} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/analyze>)
  endif()
  if(STE_ASAN)
    target_compile_options(${t} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/fsanitize=address;/Zi>)
    target_link_options(${t} PRIVATE /INCREMENTAL:NO)
  endif()
endforeach()
if(STE_ASAN)
  get_filename_component(STE_COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(STE_ASAN_ARCH x86_64)
  else()
    set(STE_ASAN_ARCH i386)
  endif()
  find_file(STE_ASAN_DLL NAMES clang_rt.asan_dynamic-${STE_ASAN_ARCH}.dll PATHS "${STE_COMPILER_DIR}" NO_DEFAULT_PATH REQUIRED)
  foreach(t service-trigger-explainer ste_tests)
    add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${STE_ASAN_DLL}" "$<TARGET_FILE_DIR:${t}>")
  endforeach()
endif()
add_test(NAME unit COMMAND ste_tests)
add_test(NAME integration COMMAND ste_tests integration)
set_tests_properties(integration PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME help COMMAND service-trigger-explainer --help)
add_test(NAME version COMMAND service-trigger-explainer --version)
add_test(NAME cli_errors COMMAND ${CMAKE_COMMAND} -DAPP=$<TARGET_FILE:service-trigger-explainer> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
add_test(NAME source_audit COMMAND ${CMAKE_COMMAND} -DROOT=${CMAKE_CURRENT_SOURCE_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/audit.cmake)
find_program(STE_PWSH pwsh)
if(STE_PWSH)
  add_test(NAME json_independent COMMAND ${STE_PWSH} -NoProfile -File ${CMAKE_CURRENT_SOURCE_DIR}/tests/json.ps1 -App $<TARGET_FILE:service-trigger-explainer> -Tests $<TARGET_FILE:ste_tests> -Schema ${CMAKE_CURRENT_SOURCE_DIR}/docs/schema-v1.json)
endif()
install(TARGETS service-trigger-explainer RUNTIME DESTINATION bin)
install(FILES README.md LICENSE VALIDATION.md DESTINATION share/service-trigger-explainer)
install(FILES docs/schema-v1.json DESTINATION share/service-trigger-explainer/docs)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "service-trigger-explainer-${PROJECT_VERSION}-windows-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
